<div class="flex justify-between items-center mb-1">
<span class="font-bold text-xs sm:text-sm text-slate-900" contenteditable="true" onblur="updateShorts(${index}, 'title', this.innerText)">${item.title}</span>
<span class="bg-red-100 text-red-800 text-[10px] font-bold px-2 py-0.5 rounded-full uppercase" contenteditable="true" onblur="updateShorts(${index}, 'tag', this.innerText)">${item.tag}</span>
</div>
<p class="text-xs text-slate-600 italic" contenteditable="true" onblur="updateShorts(${index}, 'desc', this.innerText)">${item.desc}</p>
</div>
<button onclick="deleteShorts(event, ${index})" class="no-print text-slate-300 hover:text-red-500 font-bold text-xs">✕</button>
`;
container.appendChild(div);
});
}
function toggleShorts(index) {
shortsData[index].done = !shortsData[index].done;
renderShorts();
saveState();
}
function updateShorts(index, field, value) {
shortsData[index][field] = value;
saveState();
}

}
function deleteShorts(e, index) {

shortsData.splice(index, 1);
renderShorts();
saveState();
}
